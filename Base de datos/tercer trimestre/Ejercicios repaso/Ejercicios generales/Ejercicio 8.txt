Ejercicio 8


Hacer lo mismo que el 7 pero con bloqueo FOR UPDATE.

start transaction;

select saldo
from cuentas
where id_cuenta in (1, 2)
order by id_cuenta
for update;

update cuentas
set saldo = saldo - 40
where id_cuenta = 1
and saldo >= 40;

update cuentas
set saldo = saldo + 40
where id_cuenta = 2;

commit;
