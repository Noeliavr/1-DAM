Ejercicio 10

Corregir el deadlock usando orden de bloqueo.

Sesión 1

start transaction;

select saldo
from cuentas
where id_cuenta in (1, 2)
order by id_cuenta
for update;

update cuentas
set saldo = saldo - 20
where id_cuenta = 1;

update cuentas
set saldo = saldo + 20
where id_cuenta = 2;

commit;


Sesión 2

start transaction;

select saldo
from cuentas
where id_cuenta in (1, 2)
order by id_cuenta
for update;

update cuentas
set saldo = saldo - 30
where id_cuenta = 2;

update cuentas
set saldo = saldo + 30
where id_cuenta = 1;
