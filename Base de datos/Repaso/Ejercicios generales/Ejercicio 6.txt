Ejercicio 6

Crear un trigger que impida insertar saldo negativo.

delimiter $$

drop trigger if exists trg_cliente_saldo_insert $$

create trigger trg_cliente_saldo_insert
before insert
on clientes
for each row
begin
    if new.saldo < 0 then
        signal sqlstate '45000'
        set message_text = 'El saldo no puede ser negativo';
    end if;
end $$

delimiter ;
