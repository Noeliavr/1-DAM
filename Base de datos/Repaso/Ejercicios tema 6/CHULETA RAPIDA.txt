    in p1 int,
    out res boolean,
    out msg varchar(100)
)

    declare v int default 0;

    begin
        rollback;
        set res=false;

    end;

    start transaction;

    -- lógica

    set res=true;
    set msg='ok';

🔹 CURSOR

declare fin boolean default false;

declare c cursor for
    select id from tabla;

declare continue handler for not found set fin = true;

open c;

loop1: loop
    fetch c into v_id;

    if fin then
        leave loop1;
    end if;

end loop;

close c;
