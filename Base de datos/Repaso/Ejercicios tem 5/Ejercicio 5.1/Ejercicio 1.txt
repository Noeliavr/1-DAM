Dadas las tablas:

create table clientes (
    id_cliente int primary key,
    nombre varchar(50),
    ciudad varchar(50)
);

create table pedidos (
    id_pedido int primary key,
    id_cliente int,
    total decimal(10,2),
    foreign key (id_cliente) references clientes(id_cliente)
);

Crear una tabla clientes_top e insertar en ella:

id_cliente
nombre
total_compras

Solo de aquellos clientes cuyo total de compras sea mayor que la media global.



create table clientes_top (
    id_cliente int,
    nombre varchar(50),
    total_compras decimal(10,2)
);

insert into clientes_top (id_cliente, nombre, total_compras)
select c.id_cliente, c.nombre, sum(p.total)
from clientes c
join pedidos p on c.id_cliente = p.id_cliente
group by c.id_cliente, c.nombre
having sum(p.total) > (
    select avg(total_cliente)
    from (
        select sum(total) as total_cliente
        from pedidos
        group by id_cliente
    ) t
);
