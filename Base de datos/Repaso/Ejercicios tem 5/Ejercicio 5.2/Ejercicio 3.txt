Dadas las tablas:

create table productos (
    id_producto int primary key,
    nombre varchar(50),
    stock int
);

create table ventas (
    id_venta int auto_increment primary key,
    id_producto int,
    cantidad int,
    foreign key (id_producto) references productos(id_producto)
);

insert into productos values
(1, 'Teclado', 5),
(2, 'Raton', 0);


Realizar una venta de 3 unidades del producto 1.
Debe hacerse con transacción. Si no hay stock suficiente, se debe hacer ROLLBACK.



start transaction;

update productos
set stock = stock - 3
where id_producto = 1
and stock >= 3;

insert into ventas (id_producto, cantidad)
values (1, 3);

commit;
